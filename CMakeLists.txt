cmake_minimum_required(VERSION 3.16)
project(MiniCompiler VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(minicompiler_lib
    src/lexer.cpp src/parser.cpp src/ir.cpp src/optimizer.cpp src/codegen.cpp)
target_include_directories(minicompiler_lib PUBLIC include)
if(MSVC)
  target_compile_options(minicompiler_lib PRIVATE /W4 /permissive-)
else()
  target_compile_options(minicompiler_lib PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(minicompiler src/main.cpp)
target_link_libraries(minicompiler PRIVATE minicompiler_lib)
add_executable(minicompiler_tests tests/tests.cpp)
target_link_libraries(minicompiler_tests PRIVATE minicompiler_lib)
enable_testing()
add_test(NAME minicompiler_tests COMMAND minicompiler_tests)
